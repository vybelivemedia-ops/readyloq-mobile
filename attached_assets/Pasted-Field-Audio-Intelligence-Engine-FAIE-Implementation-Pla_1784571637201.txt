processing/AudioConfidence.ts
TypeScript
export enum ConfidenceAction {
  AUTO_COMMIT = 'AUTO_COMMIT',
  CONFIRM = 'CONFIRM',
  ASK_AGAIN = 'ASK_AGAIN',
}

export class AudioConfidence {
  static evaluate(score: number): ConfidenceAction {
    if (score > 0.92) return ConfidenceAction.AUTO_COMMIT;
    if (score >= 0.70) return ConfidenceAction.CONFIRM;
    return ConfidenceAction.ASK_AGAIN;
  }
}

intelligence/SpeechIntentEngine.ts
TypeScript
import { INSPECTION_DICTIONARY } from './VocabularyRegistry';

export interface StructuredInspectionEvent {
  action: string;
  product?: string;
  room?: string;
  pest?: string;
  evidence?: string;
  confidence: number;
}

export class SpeechIntentEngine {
  parse(rawText: string, baseConfidence: number, visionContext?: any): StructuredInspectionEvent {
    let cleanedText = rawText.toLowerCase();

    // Map company aliases
    Object.entries(INSPECTION_DICTIONARY.aliases).forEach(([alias, formal]) => {
      if (cleanedText.includes(alias)) {
        cleanedText = cleanedText.replace(alias, formal.toLowerCase());
      }
    });

    // Extract intent and match against dictionary entities
    // Merge visionContext if phrases like "Seal this" or "Another bait station" are present
    return {
      action: 'applyTreatment',
      product: 'Alpine WSG',
      room: 'Kitchen',
      confidence: baseConfidence,
    };
  }
}
6. Session Orchestration & Offline Sync
AudioSessionManager.ts
TypeScript
import { SpeechProvider } from './providers/SpeechProvider';
import { NoiseSuppression } from './preprocessing/NoiseSuppression';
import { SpeechIntentEngine, StructuredInspectionEvent } from './intelligence/SpeechIntentEngine';
import { AudioConfidence, ConfidenceAction } from './processing/AudioConfidence';

export class AudioSessionManager {
  private preprocessor = new NoiseSuppression();
  private intentEngine = new SpeechIntentEngine();

  constructor(private provider: SpeechProvider) {}

  async startSession(onStateUpdate: (event: StructuredInspectionEvent, action: ConfidenceAction) => void) {
    await this.provider.start();

    this.provider.onTranscription(async (result) => {
      if (!result.isFinal) return;

      const action = AudioConfidence.evaluate(result.confidence);
      const structuredEvent = this.intentEngine.parse(result.text, result.confidence);

      onStateUpdate(structuredEvent, action);
    });
  }

  async stopSession() {
    await this.provider.stop();
  }
}
